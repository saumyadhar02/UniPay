    // --- Storage keys ---
    const USERS_KEY = 'unipay_users_v1';
    const SUBS_KEY = 'unipay_subs_v1';

    let users = {};
    let currentUser = null; // phone string

    // --- Persistence ---
    function loadUsers(){
        try{ users = JSON.parse(localStorage.getItem(USERS_KEY) || '{}'); }catch(e){ users = {}; }
        if(Object.keys(users).length===0) initDemoData();
    }
    function saveUsers(){ localStorage.setItem(USERS_KEY, JSON.stringify(users)); }

    function loadSubs(){
        try{return JSON.parse(localStorage.getItem(SUBS_KEY) || '{}');}catch(e){return {};}
    }
    function saveSubs(s){ localStorage.setItem(SUBS_KEY, JSON.stringify(s)); }

    // --- Demo data ---
    function initDemoData(){
        users = {
            '9876543210': {
                phone: '9876543210',
                name: 'Demo User',
                pin: '1234',
                walletId: 'W' + Date.now().toString().slice(-6),
                balance: 1000,
                transactions: []
            }
        };
        saveUsers();
    }

    // --- Navigation ---
    function showPage(pageId){
        document.querySelectorAll('.page').forEach(p=>p.classList.remove('active'));
        const el = document.getElementById(pageId);
        if(el) el.classList.add('active');
        // show/hide bottom nav
        document.getElementById('bottomNav').style.display = (pageId==='dashboardPage' && currentUser)?'flex':'none';
    }

    function showTab(tabId){
        document.querySelectorAll('.tab-content').forEach(t=>t.style.display='none');
        const el = document.getElementById(tabId);
        if(el) el.style.display='block';
        if(tabId==='homeTab' || tabId==='historyTab') updateTransactions();
        if(tabId==='profileTab') updateProfile();
        if(tabId==='sendTab' && currentUser){ document.getElementById('availableBalance').textContent = (users[currentUser].balance||0).toFixed(2); }
    }

    // --- Alerts ---
    function showAlert(id,msg,type){

    // --- QR scan and paste processing ---
    let qrStream = null; let qrScanInterval = null;
    async function startQRScan(){
        const video = document.getElementById('qrVideo');
        const container = document.getElementById('qrVideoContainer');
        if(!navigator.mediaDevices || !navigator.mediaDevices.getUserMedia){ showAlert('qrAlert','Camera not supported. Use paste.', 'error'); return; }
        try{
            const stream = await navigator.mediaDevices.getUserMedia({video:{facingMode:'environment'}});
            qrStream = stream; video.srcObject = stream; container.style.display='block';
            await video.play();
            if(window.BarcodeDetector){
                const detector = new BarcodeDetector({formats:['qr_code']});
                qrScanInterval = setInterval(async ()=>{
                    try{
                        const codes = await detector.detect(video);
                        if(codes && codes.length){ parseQRPayload(codes[0].rawValue); stopQRScan(); }
                    }catch(e){ /* ignore frame errors */ }
                },400);
            }else{
                // fallback: capture image frames and try to decode via canvas if needed - but we'll use paste fallback
                showAlert('qrAlert','Camera scanning started — your browser may not support native QR detection. Use paste if it fails.','success');
            }
        }catch(err){ showAlert('qrAlert','Camera access denied or error','error'); }
    }
    function stopQRScan(){ if(qrStream){ qrStream.getTracks().forEach(t=>t.stop()); qrStream=null; } const video = document.getElementById('qrVideo'); video.pause(); document.getElementById('qrVideoContainer').style.display='none'; if(qrScanInterval){ clearInterval(qrScanInterval); qrScanInterval=null; } }

    function processQRPaste(){ const payload = document.getElementById('qrPaste').value.trim(); if(!payload) return showAlert('qrAlert','Paste QR payload','error'); parseQRPayload(payload); }

    function parseQRPayload(payload){
        // support: pay:PHONE:AMOUNT
        // service:TYPE:ACCOUNT:AMOUNT
        // unipay://pay?to=...&amount=...
        let p = payload.trim();
        if(p.startsWith('unipay://')){
            try{
                const url = new URL(p);
                const to = url.searchParams.get('to'); const amount = parseFloat(url.searchParams.get('amount'));
                if(to && amount) return payToPhone(to, amount);
            }catch(e){}
        }
        const parts = p.split(':');
        if(parts[0]==='pay' && parts.length>=3){ const to=parts[1]; const amt=parseFloat(parts[2]); return payToPhone(to,amt); }
        if(parts[0]==='service' && parts.length>=4){ const svc=parts[1]; const acct=parts[2]; const amt=parseFloat(parts[3]); return payServicePayload(svc,acct,amt); }
        showAlert('qrAlert','Unsupported QR payload','error');
    }

    function payToPhone(phone, amount){ if(!currentUser) return showAlert('qrAlert','Login first','error'); const recipient = users[phone]; if(!recipient) return showAlert('qrAlert','Recipient not found','error'); if(amount<=0||amount>users[currentUser].balance) return showAlert('qrAlert','Insufficient balance','error'); users[currentUser].balance -= amount; recipient.balance = (recipient.balance||0)+amount; const ts=new Date().toISOString(); users[currentUser].transactions.push({id:Date.now(),type:'debit',amount,desc:`Sent via QR to ${phone}`,date:ts,balance:users[currentUser].balance}); recipient.transactions.push({id:Date.now()+1,type:'credit',amount,desc:`Received via QR from ${currentUser}`,date:ts,balance:recipient.balance}); saveUsers(); updateDashboard(); updateTransactions(); showAlert('qrAlert',`Sent ₹${amount} to ${phone}`,'success'); }

    function payServicePayload(service,acct,amount){ if(!currentUser) return showAlert('qrAlert','Login first','error'); if(amount<=0||amount>users[currentUser].balance) return showAlert('qrAlert','Insufficient balance','error'); users[currentUser].balance -= amount; const ts=new Date().toISOString(); users[currentUser].transactions.push({id:Date.now(),type:'debit',amount,desc:`Paid ${service} (${acct}) via QR`,date:ts,balance:users[currentUser].balance}); saveUsers(); updateDashboard(); updateTransactions(); showAlert('qrAlert',`Paid ₹${amount} for ${service}`,'success'); }

    function processDueSubscriptions(){ const subs = loadSubs(); const allPhones = Object.keys(subs); const today = new Date(); let changed=false; for(const phone of allPhones){ const arr = subs[phone]; for(const s of arr){ const next = new Date(s.nextDate + 'T00:00:00'); if(next <= today){ // attempt payment
                    const u = users[phone]; if(!u) continue; if(u.balance >= s.amount){ u.balance -= s.amount; const ts = new Date().toISOString(); u.transactions.push({id:Date.now(),type:'debit',amount:s.amount,desc:`Subscription ${s.service}`,date:ts,balance:u.balance}); // advance next by one month
                            const nd = new Date(next); nd.setMonth(nd.getMonth()+1); s.nextDate = nd.toISOString().slice(0,10); changed=true; saveUsers(); }
                }
            }
        }
        if(changed) saveSubs(subs);
    }

    // Send Money implementation
    function sendMoney(){
        if(!currentUser) return showAlert('sendAlert','Login first','error');
        const recipient = document.getElementById('recipient').value.trim();
        const amount = parseFloat(document.getElementById('sendAmount').value);
        const pin = document.getElementById('sendPin').value.trim();
        if(!recipient || !amount || !pin) return showAlert('sendAlert','Please fill all fields','error');
        const sender = users[currentUser];
        if(pin !== sender.pin) return showAlert('sendAlert','Incorrect PIN','error');
        if(amount <= 0 || amount > sender.balance) return showAlert('sendAlert','Invalid or insufficient amount','error');
        const recipientUser = users[recipient] || Object.values(users).find(u=>u.walletId===recipient);
        if(!recipientUser) return showAlert('sendAlert','Recipient not found','error');
        sender.balance -= amount;
        recipientUser.balance = (recipientUser.balance||0) + amount;
        const ts = new Date().toISOString();
        sender.transactions.push({id:Date.now(), type:'debit', amount, desc:`Sent to ${recipientUser.name||recipient}`, date:ts, balance:sender.balance});
        recipientUser.transactions.push({id:Date.now()+1, type:'credit', amount, desc:`Received from ${sender.name||currentUser}`, date:ts, balance:recipientUser.balance});
        saveUsers(); updateDashboard(); updateTransactions(); showAlert('sendAlert',`₹${amount.toFixed(2)} sent to ${recipientUser.name||recipient}`,'success');
        document.getElementById('recipient').value=''; document.getElementById('sendAmount').value=''; document.getElementById('sendPin').value='';
    }
